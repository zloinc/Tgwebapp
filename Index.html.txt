    <script>
        // Сообщаем Телеграму, что приложение готово
        const tg = window.Telegram.WebApp;
        tg.expand();
        tg.ready();

        // Логика переключения вкладок
        function showSection(id) {
            // Скрываем все секции
            document.querySelectorAll('.section').forEach(el => el.classList.remove('active'));
            // Сбрасываем цвета кнопок
            document.querySelectorAll('.nav-btn').forEach(btn => btn.classList.replace('accent-text', 'text-gray-400'));
            
            // Показываем нужную секцию
            document.getElementById(id).classList.add('active');
            // Подсвечиваем активную кнопку
            document.getElementById('btn-' + id).classList.replace('text-gray-400', 'accent-text');
            
            // Легкая вибрация при клике (работает только внутри Telegram)
            if (tg.HapticFeedback) {
                tg.HapticFeedback.impactOccurred('light');
            }
        }

        // Подсвечиваем первую кнопку при загрузке
        document.getElementById('btn-about').classList.replace('text-gray-400', 'accent-text');
    </script>
